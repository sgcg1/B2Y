# 1. 看 skopeo 装没装
which skopeo

# 2. 没装就装（root 可装，只装一个独立工具，不影响 docker 和容器）
apt-get install -y skopeo      # Ubuntu/Debian
# 或 yum install -y skopeo     # CentOS/RHEL

# 3. 走 shell 代理把镜像拉成 tar（写到你自己的目录）
cd /home/yjl
skopeo copy \
  docker://quay.io/celenewyear/tilelang-ascend:cann9.3.0-weekly.20261021-ubuntu22.04-py3.12 \
  docker-archive:/home/yjl/tilelang-ascend.tar:cann9.3.0-weekly.20261021-ubuntu22.04-py3.12

# 4. 本地导入（不联网、不重启、不影响同事）
docker load -i /home/yjl/tilelang-ascend.tar

# 5. 确认能看到完整名字
docker images | grep tilelang
